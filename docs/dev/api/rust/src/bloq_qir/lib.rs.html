1//! Standard QIR 2.1 Adaptive emission from typed Bloq VM programs.
2//!
3//! The emitter retains classical control and uses the decoder's external
4//! reset, enqueue, correction and nonblocking readiness interfaces. Device
5//! durations are assigned downstream; nonzero source releases and explicit
6//! idle intervals are rejected. LLVM 21 tools promote private construction
7//! registers to SSA and verify the final module before returning text/bitcode.

11mod validation;

17/// Binding of one observable to a configured syndrome decoder.
18#[derive(Debug, Clone)]
19pub struct DecoderBinding {
20    /// Observable task whose Corrected and Flip requests share a solve.
21    pub observable: TaskId,
22    /// Runtime decoder identifier, independent of the emitter's name.
23    pub decoder: u32,
24    /// Ordered syndrome bits, packed least-significant bit first.
25    pub syndrome: Box<[RecordParity]>,
26    /// Width of the configured decoder's correction mask (1–64).
27    pub correction_count: u32,
28    /// Correction-mask bit for this observable.
29    pub correction_bit: u32,
30}

32/// Options for standard Adaptive emission.
33#[derive(Debug, Clone)]
34pub struct QirOptions {
35    /// Explicit model/window bindings; the VM's stochastic mock is not exported.
36    pub decoders: Vec<DecoderBinding>,
37    /// Terminal classical bit outputs, in recording order.
38    pub output_bits: Vec<BitId>,
39    /// Terminal measurement outputs, captured before result-slot reuse.
40    pub output_records: Vec<RecordId>,
41    /// Bound on source-isolated retries; exhaustion returns exit code 1.
42    pub max_attempts: u32,
43    /// Bound on protection rounds; exhaustion returns exit code 2.
44    pub max_wait_rounds: u32,
45    /// LLVM 21 optimizer used for SSA promotion and module verification.
46    pub optimizer: PathBuf,
47    /// LLVM 21 assembler used for verified bitcode output.
48    pub assembler: PathBuf,
49}
50
51impl Default for QirOptions {
52    fn default() -> Self {
53        Self {
54            decoders: Vec::new(),
55            output_bits: Vec::new(),
56            output_records: Vec::new(),
57            max_attempts: 100,
58            max_wait_rounds: 100_000,
59            optimizer: "opt-21".into(),
60            assembler: "llvm-as-21".into(),
61        }
62    }
63}

65/// Verified standard Adaptive module and its resource requirements.
66#[derive(Debug, Clone)]
67pub struct QirArtifact {
68    /// LLVM text with SSA classical values and QIR 2.1 flags.
69    pub llvm_ir: String,
70    /// LLVM bitcode of the same verified module.
71    pub bitcode: Vec<u8>,
72    /// Physical resources including the QND measurement scratch ancilla.
73    pub qubit_count: u32,
74    /// Statically allocated measurement sites, including scratch results.
75    pub result_count: u32,
76}

78/// Failures at the typed VM-to-QIR boundary.
79#[derive(Debug, thiserror::Error)]
80pub enum QirEmissionError {
81    /// A malformed identifier, value requirement or control structure.
82    #[error("invalid QIR input: {0}")]
83    InvalidProgram(String),
84    /// A source behavior lacks an implemented lowering contract.
85    #[error("unsupported QIR behavior: {0}")]
86    Unsupported(String),
87    /// A native LLVM tool could not be started or communicated with.
88    #[error("LLVM tool I/O: {0}")]
89    ToolIo(#[from] std::io::Error),
90    /// LLVM rejected the generated module.
91    #[error("LLVM tool failed: {0}")]
92    Llvm(String),
93}

95/// Emit and verify standard QIR 2.1 Adaptive text and bitcode.
96///
97/// Input/output patch initialization must be authored in the physical program;
98/// simulator postselection and cloned-state probes are not exported. Bindings
99/// name decoder model outputs rather than inheriting VM mock decisions.
100///
101/// # Errors
102///
103/// Rejects malformed programs, unsupported timing/noise/patch I/O, missing
104/// decoder bindings, and native tool or LLVM verification failures.
105pub fn emit_program_qir(
106    program: &bloq_vm::Program,
107    options: &QirOptions,
108) -> Result<QirArtifact, QirEmissionError> {
109    emit::emit(program, options)
110}
